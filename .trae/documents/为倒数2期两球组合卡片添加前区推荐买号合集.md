### 3.2 JavaScript函数
```javascript
// 更新倒数2期两球组合平均率最高统计期详情的推荐买号合集
function updateBuyNumbersCollectionSecondLast() {
  const tbody = document.getElementById('detailResultsSecondLast');
  const contentDiv = document.getElementById('buyNumbersCollectionContentSecondLast');
  
  if (!tbody || !contentDiv) {
    console.error('找不到tbody或contentDiv元素');
    return;
  }
  
  // 使用Set收集所有推荐买号，自动去重
  const buyNumbersSet = new Set();
  
  // 获取所有行
  const rows = tbody.querySelectorAll('tr');
  
  // 遍历所有行，收集推荐买号
  rows.forEach(row => {
    // 获取所有单元格
    const cells = row.querySelectorAll('td');
    
    // 检查是否有足够的单元格
    if (cells.length >= 8) {
      // 获取第7个单元格（下下期前区推荐买号）
      const buyNumbersCell = cells[6];
      
      // 方法1：从.red-ball元素中提取号码
      const redBalls = buyNumbersCell.querySelectorAll('.red-ball');
      let hasRedBalls = false;
      
      if (redBalls.length > 0) {
        hasRedBalls = true;
        // 收集所有红色球的号码
        redBalls.forEach(ball => {
          const ballNumber = ball.textContent.trim();
          // 只添加非空号码
          if (ballNumber) {
            buyNumbersSet.add(parseInt(ballNumber));
          }
        });
      }
      
      // 方法2：如果没有red-ball元素，直接从单元格文本中提取号码
      if (!hasRedBalls) {
        // 直接从单元格文本中提取号码
        const cellText = buyNumbersCell.textContent.trim();
        
        // 使用正则表达式提取所有数字
        const numbers = cellText.match(/\d+/g) || [];
        
        // 将提取到的数字添加到Set中
        numbers.forEach(num => {
          buyNumbersSet.add(parseInt(num));
        });
      }
    }
  });
  
  // 如果没有推荐买号数据
  if (buyNumbersSet.size === 0) {
    contentDiv.innerHTML = '<div style="color: #666;">暂无推荐买号数据</div>';
    return;
  }
  
  // 将Set转换为数组，排序后生成HTML
  const buyNumbersArray = Array.from(buyNumbersSet).sort((a, b) => a - b);
  let html = '';
  
  // 生成球的HTML
  buyNumbersArray.forEach(number => {
    html += `<span class="red-ball">${number}</span>`;
  });
  
  // 更新contentDiv的内容
  contentDiv.innerHTML = html;
}
```
