**关键代码变更：**
为每个performSearchX函数添加以下逻辑：
```javascript
// 按回测方法和平均正确率分组
const groupedResults = {};

allResults.forEach(result => {
  // 创建分组键：回测方法 + 平均正确率
  const key = `${result.backtestMethod}_${result.accuracy}`;
  
  if (!groupedResults[key]) {
    groupedResults[key] = [];
  }
  
  groupedResults[key].push(result);
});

// 处理每个分组
const filteredResults = [];

for (const key in groupedResults) {
  const group = groupedResults[key];
  
  // 如果分组只有一个结果，直接添加
  if (group.length === 1) {
    filteredResults.push(group[0]);
    continue;
  }
  
  // 否则，检查推荐买号是否相同
  const buyNumbersMap = new Map();
  
  group.forEach(result => {
    // 将推荐买号转换为字符串作为键，方便比较
    const buyNumbersKey = result.backBuyNumbers.sort().join(',');
    
    // 每个推荐买号组合只保留一个结果
    if (!buyNumbersMap.has(buyNumbersKey)) {
      buyNumbersMap.set(buyNumbersKey, result);
    }
  });
  
  // 将唯一的推荐买号结果添加到最终结果中
  filteredResults.push(...buyNumbersMap.values());
}

// 渲染筛选后的结果
await renderDetailDataX(filteredResults, nextPeriod);
```
