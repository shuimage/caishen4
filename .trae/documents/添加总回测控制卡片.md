```javascript
// 总回测函数
async function performAllBacktests() {
  // 显示总进度
  const totalProgressDiv = document.getElementById('totalProgress');
  const totalProgressText = document.getElementById('totalProgressText');
  const totalProgressBar = document.getElementById('totalProgressBar');
  const totalProgressPercentage = document.getElementById('totalProgressPercentage');
  const totalCurrentPhase = document.getElementById('totalCurrentPhase');
  
  totalProgressDiv.style.display = 'block';
  totalProgressText.textContent = '开始所有回测...';
  totalProgressBar.style.width = '0%';
  totalProgressPercentage.textContent = '0%';
  totalCurrentPhase.textContent = '正在准备...';
  
  // 显示/隐藏按钮
  document.getElementById('startAllBacktestBtn').style.display = 'none';
  document.getElementById('stopAllBacktestBtn').style.display = 'inline-block';
  
  // 回测函数数组
  const backtestFunctions = [
    { name: '最新1期两球组合', func: performSearch },
    { name: '倒数2期三球组合', func: performSearch2 },
    { name: '倒数3期三球组合', func: performSearch3 },
    { name: '倒数4期两球组合', func: performSearch4 },
    { name: '倒数5期两球组合', func: performSearch5 }
  ];
  
  // 执行所有回测
  for (let i = 0; i < backtestFunctions.length; i++) {
    const backtest = backtestFunctions[i];
    
    // 更新总进度
    const progress = ((i + 1) / backtestFunctions.length) * 100;
    totalProgressText.textContent = `正在执行${backtest.name}回测...`;
    totalProgressBar.style.width = `${progress}%`;
    totalProgressPercentage.textContent = `${Math.round(progress)}%`;
    totalCurrentPhase.textContent = `执行回测：${i + 1}/${backtestFunctions.length}`;
    
    // 执行回测
    await backtest.func();
  }
  
  // 更新总进度为完成
  totalProgressText.textContent = '所有回测完成！';
  totalProgressBar.style.width = '100%';
  totalProgressPercentage.textContent = '100%';
  totalCurrentPhase.textContent = '回测完成';
  
  // 显示/隐藏按钮
  document.getElementById('startAllBacktestBtn').style.display = 'inline-block';
  document.getElementById('stopAllBacktestBtn').style.display = 'none';
  
  // 隐藏总进度
  setTimeout(() => {
    totalProgressDiv.style.display = 'none';
  }, 2000);
}
```

```javascript
// 复制所有推荐买号
async function copyAllBuyNumbers() {
  try {
    // 收集所有推荐买号
    let allBuyNumbers = [];
    
    // 获取每个表格的推荐买号
    const tables = document.querySelectorAll('.detail-list');
    tables.forEach(table => {
      const rows = table.querySelectorAll('tbody tr:not(:first-child)');
      rows.forEach(row => {
        const buyNumbersCells = row.querySelectorAll('td:nth-child(6)');
        buyNumbersCells.forEach(cell => {
          const ballElements = cell.querySelectorAll('.red-ball');
          const numbers = Array.from(ballElements).map(ball => ball.textContent);
          if (numbers.length > 0) {
            allBuyNumbers.push(numbers);
          }
        });
      });
    });
    
    // 格式化并复制到剪贴板
    const formattedNumbers = allBuyNumbers.map(numbers => numbers.join('\t')).join('\n');
    await navigator.clipboard.writeText(formattedNumbers);
    alert('所有推荐买号已复制到剪贴板！');
  } catch (error) {
    console.error('复制推荐买号失败:', error);
    alert('复制推荐买号失败，请重试');
  }
}
```

```javascript
// 保存回测结果
async function saveBacktestResults() {
  try {
    // 收集所有回测结果
    const results = [];
    
    // 获取每个表格的结果
    const tables = document.querySelectorAll('.detail-list');
    tables.forEach((table, tableIndex) => {
      const header = table.closest('.section').querySelector('h3').textContent;
      const rows = table.querySelectorAll('tbody tr:not(:first-child)');
      
      const tableResults = Array.from(rows).map(row => {
        const cells = row.querySelectorAll('td');
        return {
          method: cells[0]?.textContent || '',
          backtestPeriod: cells[1]?.textContent || '',
          statsPeriod: cells[2]?.textContent || '',
          accuracy: cells[3]?.textContent || '',
          currentIssue: cells[4]?.textContent || '',
          recommendedNumbers: Array.from(cells[5]?.querySelectorAll('.red-ball') || []).map(ball => ball.textContent).join(','),
          operation: cells[6]?.innerHTML || ''
        };
      });
      
      results.push({
        title: header,
        data: tableResults
      });
    });
    
    // 保存到localStorage
    localStorage.setItem('backtestResults', JSON.stringify(results));
    alert('回测结果保存成功！');
  } catch (error) {
    console.error('保存回测结果失败:', error);
    alert('保存回测结果失败，请重试');
  }
}
```

#### 3.4 加载回测结果函数
```javascript
// 加载回测结果
async function loadBacktestResults() {
  try {
    // 从localStorage获取结果
    const savedResults = localStorage.getItem('backtestResults');
    if (!savedResults) {
      alert('没有找到保存的回测结果！');
      return;
    }
    
    const results = JSON.parse(savedResults);
    
    // 加载结果到对应表格
    const tables = document.querySelectorAll('.detail-list');
    results.forEach((result, index) => {
      if (tables[index]) {
        const tbody = tables[index].querySelector('tbody');
        const header = tables[index].closest('.section').querySelector('h3').textContent;
        
        // 确保标题匹配
        if (header === result.title) {
          // 清空现有内容
          tbody.innerHTML = '';
          
          // 添加加载的结果
          result.data.forEach(rowData => {
            const tr = document.createElement('tr');
            tr.innerHTML = `
              <td>${rowData.method}</td>
              <td>${rowData.backtestPeriod}</td>
              <td>${rowData.statsPeriod}</td>
              <td>${rowData.accuracy}</td>
              <td>${rowData.currentIssue}</td>
              <td>${rowData.recommendedNumbers.split(',').map(num => `<span class="red-ball">${num}</span>`).join(' ')}</td>
              <td>${rowData.operation}</td>
            `;
            tbody.appendChild(tr);
          });
        }
      }
    });
    
    alert('回测结果加载成功！');
  } catch (error) {
    console.error('加载回测结果失败:', error);
    alert('加载回测结果失败，请重试');
  }
}
```
