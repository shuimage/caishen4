## 1. 两球/三球组合算法

### 1.1 组合生成算法

从5个前区号码中生成所有两球组合（C(5,2) = 10个）：

```javascript
function generateCombinations(numbers) {
  const combinations = [];
  for (let i = 0; i < numbers.length; i++) {
    for (let j = i + 1; j < numbers.length; j++) {
      // 确保小球在前，大球在后
      const ball1 = Math.min(numbers[i], numbers[j]);
      const ball2 = Math.max(numbers[i], numbers[j]);
      combinations.push(`${ball1}-${ball2}`);
    }
  }
  return combinations;
}

// 示例：前区号码 [3, 8, 15, 24, 33]
// 生成组合：["3-8", "3-15", "3-24", "3-33", "8-15", "8-24", "8-33", "15-24", "15-33", "24-33"]
```

### 1.2 三球组合生成

从5个前区号码中生成所有三球组合（C(5,3) = 10个）：

```javascript
function generateTripleCombinations(numbers) {
  const combinations = [];
  for (let i = 0; i < numbers.length; i++) {
    for (let j = i + 1; j < numbers.length; j++) {
      for (let k = j + 1; k < numbers.length; k++) {
        const sorted = [numbers[i], numbers[j], numbers[k]].sort((a, b) => a - b);
        combinations.push(`${sorted[0]}-${sorted[1]}-${sorted[2]}`);
      }
    }
  }
  return combinations;
}
```

### 1.3 历史数据统计

统计每个组合在历史数据中的出现情况：

```javascript
function analyzeCombinations(latestDraw, historyData, period) {
  // 1. 生成最新一期的所有组合
  const combinations = generateCombinations(latestDraw.frontNumbers);
  
  // 2. 初始化统计对象
  const stats = {};
  combinations.forEach(combo => {
    stats[combo] = {
      combination: combo,
      count: 0,
      nextDrawNumbers: [],  // 记录组合出现时下一期的号码
      numberCounts: {}      // 统计每个号码出现的次数
    };
  });
  
  // 3. 遍历历史数据
  // 注意：数据按 issue 降序排列，所以下一期的索引是 i-1
  for (let i = 1; i < historyData.length; i++) {
    const currentDraw = historyData[i];    // 当前期
    const nextDraw = historyData[i - 1];   // 下一期
    
    // 生成当前期的所有组合
    const currentCombos = generateCombinations(currentDraw.frontNumbers);
    
    // 检查是否包含我们关心的组合
    currentCombos.forEach(combo => {
      if (combinations.includes(combo)) {
        // 记录下一期的号码
        nextDraw.frontNumbers.forEach(num => {
          stats[combo].nextDrawNumbers.push(num);
          stats[combo].numberCounts[num] = (stats[combo].numberCounts[num] || 0) + 1;
        });
        stats[combo].count++;
      }
    });
  }
  
  return stats;
}
```

## 2. 回测准确率计算方法

### 2.1 杀号回测算法

验证杀号策略的历史准确率：

```javascript
function backtestKillNumbers(historyData, method, period) {
  const results = [];
  
  // 对每一期进行回测（跳过最新一期，因为没有下一期数据）
  for (let i = 1; i < historyData.length; i++) {
    const currentDraw = historyData[i];    // 当前期
    const nextDraw = historyData[i - 1];   // 下一期（实际开奖）
    
    // 1. 根据当前期计算推荐杀号
    const recommendedKillNumbers = calculateKillNumbers(
      currentDraw, 
      historyData.slice(i), 
      method
    );
    
    // 2. 验证杀号准确性
    const correctKills = recommendedKillNumbers.filter(num => 
      !nextDraw.frontNumbers.includes(num)
    ).length;
    
    const wrongKills = recommendedKillNumbers.filter(num => 
      nextDraw.frontNumbers.includes(num)
    ).length;
    
    // 3. 记录结果
    results.push({
      period: currentDraw.issue,
      nextPeriod: nextDraw.issue,
      recommendedKillNumbers,
      actualNumbers: nextDraw.frontNumbers,
      correctKillCount: correctKills,
      wrongKillCount: wrongKills,
      accuracy: correctKills / recommendedKillNumbers.length
    });
  }
  
  return results;
}
```

### 2.2 杀号计算方法

根据统计方法计算推荐杀号：

```javascript
function calculateKillNumbers(currentDraw, historyData, method) {
  // 生成当前期的所有组合
  const combinations = generateCombinations(currentDraw.frontNumbers);
  
  // 统计每个号码在组合出现时的出现次数
  const numberCounts = {};
  for (let i = 1; i <= 35; i++) numberCounts[i] = 0;
  
  // 遍历历史数据
  for (let i = 2; i < historyData.length; i++) {
    const draw = historyData[i];
    const nextNextDraw = historyData[i - 2];  // 下下期
    
    const drawCombos = generateCombinations(draw.frontNumbers);
    
    drawCombos.forEach(combo => {
      if (combinations.includes(combo)) {
        nextNextDraw.frontNumbers.forEach(num => {
          numberCounts[num]++;
        });
      }
    });
  }
  
  // 根据方法选择杀号
  let killNumbers = [];
  
  switch (method) {
    case 'most':
      // 杀出现次数最多的号码（认为它们不会连续出现）
      const maxCount = Math.max(...Object.values(numberCounts));
      killNumbers = Object.entries(numberCounts)
        .filter(([num, count]) => count === maxCount && count > 0)
        .map(([num]) => parseInt(num));
      break;
      
    case 'least':
      // 杀出现次数最少的号码
      const minCount = Math.min(...Object.values(numberCounts).filter(c => c > 0));
      killNumbers = Object.entries(numberCounts)
        .filter(([num, count]) => count === minCount && count > 0)
        .map(([num]) => parseInt(num));
      break;
      
    case 'average':
      // 杀接近平均出现次数的号码
      const avg = Object.values(numberCounts).reduce((a, b) => a + b, 0) / 35;
      killNumbers = Object.entries(numberCounts)
        .filter(([num, count]) => Math.abs(count - avg) < 1)
        .map(([num]) => parseInt(num));
      break;
      
    case 'zero':
      // 推荐杀出现次数为0的号码
      killNumbers = Object.entries(numberCounts)
        .filter(([num, count]) => count === 0)
        .map(([num]) => parseInt(num));
      break;
  }
  
  return killNumbers;
}
```

## 3. 多阶段搜索策略详解

### 3.1 深度预测工作流程

深度预测通过多阶段搜索寻找最优统计期数：

```
阶段1: 大范围搜索（步长200）
  └─ 测试期数: 200, 400, 600, 800, 1000...
  └─ 找出准确率最高的区间

阶段2: 中范围搜索（步长100）
  └─ 在阶段1最优值附近 ±200 范围内
  └─ 测试期数: 最优值-200, 最优值-100, 最优值, 最优值+100, 最优值+200

阶段3: 精细搜索（步长50）
  └─ 在阶段2最优值附近 ±100 范围内
  └─ 测试期数: 最优值-100, 最优值-50, 最优值, 最优值+50, 最优值+100

最终结果: 最优统计期数 + 对应推荐号码
```

### 3.2 实现代码

```javascript
async function deepPrediction(period, zone = 'front') {
  const results = [];
  
  // 阶段1: 大范围搜索，步长200
  const stage1Periods = [200, 400, 600, 800, 1000, 1200, 1400, 1600];
  const stage1Results = [];
  
  for (const testPeriod of stage1Periods) {
    const accuracy = await testPredictionAccuracy(period, testPeriod, zone);
    stage1Results.push({ period: testPeriod, accuracy });
  }
  
  // 找出阶段1最优值
  const stage1Best = stage1Results.reduce((best, current) => 
    current.accuracy > best.accuracy ? current : best
  );
  
  // 阶段2: 中范围搜索，步长100
  const stage2Base = stage1Best.period;
  const stage2Periods = [
    Math.max(50, stage2Base - 200),
    Math.max(50, stage2Base - 100),
    stage2Base,
    stage2Base + 100,
    stage2Base + 200
  ];
  
  const stage2Results = [];
  for (const testPeriod of stage2Periods) {
    const accuracy = await testPredictionAccuracy(period, testPeriod, zone);
    stage2Results.push({ period: testPeriod, accuracy });
  }
  
  const stage2Best = stage2Results.reduce((best, current) => 
    current.accuracy > best.accuracy ? current : best
  );
  
  // 阶段3: 精细搜索，步长50
  const stage3Base = stage2Best.period;
  const stage3Periods = [
    Math.max(50, stage3Base - 100),
    Math.max(50, stage3Base - 50),
    stage3Base,
    stage3Base + 50,
    stage3Base + 100
  ];
  
  const stage3Results = [];
  for (const testPeriod of stage3Periods) {
    const accuracy = await testPredictionAccuracy(period, testPeriod, zone);
    stage3Results.push({ period: testPeriod, accuracy });
  }
  
  const stage3Best = stage3Results.reduce((best, current) => 
    current.accuracy > best.accuracy ? current : best
  );
  
  // 返回最优结果
  return {
    optimalPeriod: stage3Best.period,
    optimalAccuracy: stage3Best.accuracy,
    recommendedNumbers: await getRecommendedNumbers(period, stage3Best.period, zone),
    searchProcess: {
      stage1: stage1Results,
      stage2: stage2Results,
      stage3: stage3Results
    }
  };
}
```

### 3.3 准确率测试

```javascript
async function testPredictionAccuracy(targetPeriod, testPeriod, zone) {
  // 获取历史数据
  const historyData = await getHistoryData(testPeriod + 10);
  
  let correctCount = 0;
  let totalCount = 0;
  
  // 对历史数据进行回测
  for (let i = testPeriod; i < historyData.length - 1; i++) {
    const currentDraw = historyData[i];
    const nextDraw = historyData[i - 1];
    
    // 使用 testPeriod 期数据统计
    const statsData = historyData.slice(i, i + testPeriod);
    
    // 计算推荐号码
    const recommended = calculateRecommendedNumbers(currentDraw, statsData, zone);
    
    // 验证准确性
    const actualNumbers = zone === 'front' ? nextDraw.frontNumbers : nextDraw.backNumbers;
    const correct = recommended.filter(num => actualNumbers.includes(num)).length;
    
    correctCount += correct;
    totalCount += recommended.length;
  }
  
  return totalCount > 0 ? correctCount / totalCount : 0;
}
```

## 4. 幻圆回测原理

### 4.1 镜像对称杀号法

基于号码的镜像关系（36 - num）进行杀号：

```
前区号码镜像关系：
01 ↔ 35    02 ↔ 34    03 ↔ 33    ...    17 ↔ 19
(18 是中心号，镜像为自己)

示例：
如果当前期出现 01，则杀其镜像号 35
如果当前期出现 05，则杀其镜像号 31
```

```javascript
function mirrorSymmetryKill(currentNumbers, historyData) {
  const killNumbers = [];
  
  currentNumbers.forEach(num => {
    const mirror = 36 - num;  // 计算镜像号
    
    // 获取镜像号的冷热数据
    const hotColdInfo = calculateHotColdData(mirror, historyData);
    
    // 如果镜像号遗漏超过15期或连续出现3期以上，则杀掉
    if (hotColdInfo.missing > 15 || hotColdInfo.consecutive >= 3) {
      killNumbers.push(mirror);
    }
  });
  
  return [...new Set(killNumbers)];  // 去重
}
```

### 4.5 冷热对称杀号法

基于号码的冷热状态进行对称分析：

```javascript
function hotColdSymmetryKill(currentNumbers, historyData) {
  const killNumbers = [];
  
  // 计算所有号码的冷热数据
  const hotColdData = {};
  for (let i = 1; i <= 35; i++) {
    hotColdData[i] = calculateHotColdData(i, historyData);
  }
  
  currentNumbers.forEach(num => {
    const mirror = 36 - num;
    const numData = hotColdData[num];
    const mirrorData = hotColdData[mirror];
    
    // 如果当前号很热而镜像号很冷，或反之，则杀掉镜像号
    if ((numData.consecutive > 0 && mirrorData.missing > 10) ||
        (numData.missing > 10 && mirrorData.consecutive > 0)) {
      killNumbers.push(mirror);
    }
  });
  
  return killNumbers;
}
```

### 4.6 冷热数据计算

```javascript
function calculateHotColdData(number, historyData) {
  let consecutive = 0;  // 连续出现期数
  let missing = 0;      // 遗漏期数
  let found = false;
  
  // 计算连续出现期数
  for (let i = 0; i < historyData.length; i++) {
    const draw = historyData[i];
    if (draw.frontNumbers.includes(number)) {
      consecutive++;
    } else {
      break;
    }
  }
  
  // 计算遗漏期数
  for (let i = 0; i < historyData.length; i++) {
    const draw = historyData[i];
    if (draw.frontNumbers.includes(number)) {
      found = true;
      break;
    }
    missing++;
  }
  
  return {
    consecutive,
    missing: found ? missing : 999,  // 如果从未出现，遗漏设为极大值
    isHot: consecutive >= 2,         // 连续出现2期以上为热号
    isCold: missing > 10             // 遗漏10期以上为冷号
  };
}
```

## 5. 辅助算法

### 5.1 号码解析算法

处理数据库中不同格式的号码数据：

```javascript
function extractNumbers(data) {
  if (!data) return [];
  
  const dataStr = String(data);
  
  // 处理 JSON 数组格式: "[1, 2, 3, 4, 5]"
  if (dataStr.startsWith('[') && dataStr.endsWith(']')) {
    try {
      return JSON.parse(dataStr);
    } catch (e) {
      // 解析失败，继续其他方式
    }
  }
  
  // 使用正则提取所有数字
  const numbers = dataStr.match(/\d+/g) || [];
  return numbers.map(n => parseInt(n));
}
```

function calculateSpan(balls) {
  const nums = balls.map(ball => parseInt(ball));
  const min = Math.min(...nums);
  const max = Math.max(...nums);
  return max - min;
}

function calculateIntervalRatio(balls) {
  const nums = balls.map(ball => parseInt(ball));
  const interval1 = nums.filter(num => num <= 12).length;      // 一区: 1-12
  const interval2 = nums.filter(num => num >= 13 && num <= 24).length;  // 二区: 13-24
  const interval3 = nums.filter(num => num >= 25).length;      // 三区: 25-35
  return `${interval1}:${interval2}:${interval3}`;
}

function calculateParityRatio(balls) {
  const nums = balls.map(ball => parseInt(ball));
  const odd = nums.filter(num => num % 2 === 1).length;
  const even = nums.filter(num => num % 2 === 0).length;
