## 动态路由加载机制

### 原理

`server.js` 实现了自动路由加载机制：

```javascript
// 获取所有路由文件
const routesPath = __dirname;
const jsFiles = fs.readdirSync(routesPath).filter(f => f.endsWith('.js') && f !== 'server.js');

// 动态加载所有路由
jsFiles.forEach(file => {
  try {
    const route = require(path.join(routesPath, file));
    if (route && (route.get || route.post || route.router)) {
      // 文件名转路由路径
      const routeName = file.replace('.js', '').replace(/_/g, '/');
      app.use(`/${routeName}`, route);
    }
  } catch (err) {
    console.log(`  ❌ 加载失败: ${file} - ${err.message}`);
  }
});
```

### 路由命名规则

| 文件名 | 路由路径 | 说明 |
|--------|----------|------|
| `获取SQL数据.js` | `/huo_qu_sql_shu_ju` | 下划线转斜杠 |
| `SQL最新一期.js` | `/SQLZuiXinYiQi` | 驼峰命名保留 |
| `倒数2期两球组合.js` | `/dao_shu_2_qi_liang_qiu_zu_he` | 中文转拼音+下划线 |
| `qian_qu_shen_du_yu_ce.js` | `/qian_qu_shen_du_yu_ce` | 保持原样 |

### 路由模块规范

每个路由文件必须导出 Express Router 对象：

```javascript
const express = require('express');
const router = express.Router();

// 定义路由
router.get('/', async (req, res) => {
  // 处理逻辑
});

router.post('/some_endpoint', async (req, res) => {
  // 处理逻辑
});

module.exports = router;
```

### 示例：获取两球组合数据

```javascript
// 前端请求
fetch('http://localhost:18889/dao_shu_2_qi_liang_qiu_zu_he?period=100')
  .then(res => res.json())
  .then(data => {
    // 渲染表格
  });

// 后端处理流程
// 1. 获取倒数第2期开奖数据
// 2. 生成所有两球组合 (C(5,2)=10个)

// 4. 计算推荐杀号
// 5. 返回 JSON 结果
```

## 数据库连接池

### 配置

```javascript
// 数据库配置.js
const dbConfig = {
  host: process.env.DB_HOST || '127.0.0.1',
  user: process.env.DB_USER || 'root',
  password: process.env.DB_PASSWORD || '123456',
  database: process.env.DB_NAME || 'letou',
  port: process.env.DB_PORT || 3306,
  waitForConnections: true,
  connectionLimit: process.env.DB_POOL_LIMIT || 10,
  queueLimit: 0
};

const pool = mysql.createPool(dbConfig);
```

### 使用方式

```javascript
const { query, transaction } = require('./数据库配置.js');

// 简单查询
const rows = await query('SELECT * FROM lottery_results WHERE issue = ?', [issue]);

// 事务处理
await transaction([
  { sql: 'INSERT INTO ...', params: [...] },
  { sql: 'UPDATE ...', params: [...] }
]);
```
